# 服务费模式关联自动同步 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Link banquet purchase auto-sync and income auto-generation to the service fee mode switch, with daily cloud function to backfill missing records.

**Architecture:** New timer-triggered cloud function `autoSyncReservation` runs at 23:00 daily to backfill today+past 3 days. `reservation-add` handles instant sync on reservation create/edit with date-based gating. `purchase-add` gets a reservation picker for manual banquet purchases. Cancel/delete cleans all linked records.

**Tech Stack:** WeChat Cloud Functions, Mini Program, Cloud Database (purchase/income/reservation/settings collections)

**Files:**

```javascript
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command

/**
 * Daily 23:00 timer — backfill banquet purchase & income for today+past 3 days.
 * Skips reservations that already have linked records (manual or auto).
 */
exports.main = async (event, context) => {
  try {
    // 1. Load settings
    const settingsRes = await db.collection('settings').get()
    const settings = {}
    ;(settingsRes.data || []).forEach(s => { settings[s.key] = s.value })

    if (!settings.serviceChargeEnabled) {
      console.log('[autoSync] 服务费模式未开启，跳过')
      return { skipped: true, reason: 'serviceCharge disabled' }
    }

    const enabledDate = settings.serviceChargeEnabledDate
    const chargeNoon = Number(settings.serviceChargeNoon) || 0
    const chargeNight = Number(settings.serviceChargeNight) || 0

    // 2. Calculate date range: today-3 to today
    const now = new Date()
    const today = now.getFullYear() + '-' +
      String(now.getMonth() + 1).padStart(2, '0') + '-' +
      String(now.getDate()).padStart(2, '0')

    const past = new Date(now)
    past.setDate(past.getDate() - 3)
    const threeDaysAgo = past.getFullYear() + '-' +
      String(past.getMonth() + 1).padStart(2, '0') + '-' +
      String(past.getDate()).padStart(2, '0')

    // 3. Find eligible reservations
    const startDate = new Date(threeDaysAgo + 'T00:00:00')
    const endDate = new Date(today + 'T23:59:59')

    const resvRes = await db.collection('reservation')
      .where({
        date: _.gte(startDate).and(_.lte(endDate)),
        status: 'confirmed',
        dishPrice: _.gt(0)
      })
      .get()

    const reservations = resvRes.data || []
    console.log('[autoSync] 找到 ' + reservations.length + ' 条待处理预约')

    let created = 0, skipped = 0

    for (const r of reservations) {
      const resDate = typeof r.date === 'object'
        ? r.date.getFullYear() + '-' + String(r.date.getMonth() + 1).padStart(2, '0') + '-' + String(r.date.getDate()).padStart(2, '0')
        : r.date

      // 4. Check serviceChargeEnabledDate
      if (resDate < enabledDate) {
        skipped++
        continue
      }

      const time = r.time || '中午'
      const charge = time === '中午' ? chargeNoon : chargeNight
      const dishPrice = Number(r.dishPrice) || 0

      // 5. Generate purchase (skip if any banquet purchase already linked)
      const purchaseRes = await db.collection('purchase')
        .where({
          sourceReservationId: r._id,
          category: 'banquet'
        })
        .get()

      if (!purchaseRes.data || purchaseRes.data.length === 0) {
        const customerName = r.customerName || ''
        const roomName = r.roomName || ''
        const remark = customerName + ' - ' + roomName

        await db.collection('purchase').add({
          data: {
            amount: dishPrice,
            category: 'banquet',
            date: resDate,
            remark: remark,
            item: '',
            sourceReservationId: r._id,
            autoGenerated: true,
            createdTime: db.serverDate()
          }
        })
        created++
      }

      // 6. Generate income (skip if any income already linked)
      const incomeRes = await db.collection('income')
        .where({ reservationId: r._id })
        .get()

      if (!incomeRes.data || incomeRes.data.length === 0) {
        const amount = dishPrice + charge

        await db.collection('income').add({
          data: {
            type: 'dining',
            amount: amount,
            date: resDate,
            source: r.customerName || '',
            reservationId: r._id,
            remark: '',
            calcMode: 'dishPrice',
            dishPrice: dishPrice,
            serviceCharge: charge,
            guestCount: r.guestCount || 0,
            standard: r.standard || 0,
            roomName: r.roomName || '',
            autoGenerated: true,
            createdTime: db.serverDate()
          }
        })

        // Mark hasIncome on reservation
        await db.collection('reservation').doc(r._id).update({
          data: { hasIncome: true }
        })
        created++
      }
    }

    console.log('[autoSync] 完成: created=' + created + ', skipped=' + skipped)
    return { created, skipped }
  } catch (err) {
    console.error('[autoSync] 错误:', err)
    return { error: err.message }
  }
}
```

```javascript
async syncIncome(docData, reservationId, isCreate) {
  try {
    const dishPrice = Number(docData.dishPrice) || 0
    if (dishPrice <= 0) return
    const time = docData.time || '中午'

    // Load service charge settings
    const settingsRes = await db.queryAll(COLLECTIONS.SETTINGS, {})
    const settings = {}
    ;(settingsRes.data || []).forEach(s => { settings[s.key] = s.value })
    const charge = time === '中午'
      ? (Number(settings.serviceChargeNoon) || 0)
      : (Number(settings.serviceChargeNight) || 0)
    const amount = dishPrice + charge

    const existing = await db.queryAll(COLLECTIONS.INCOME, { reservationId })
    const hasExisting = existing.data && existing.data.length > 0
    const first = hasExisting ? existing.data[0] : null

    const app = getApp()
    const userInfo = app.globalData.userInfo || {}

    const incomeData = {
      type: 'dining',
      amount,
      date: formatDate(docData.date),
      source: docData.customerName || '',
      reservationId,
      remark: '',
      collectedBy: userInfo._id || '',
      collectedByName: userInfo.name || '',
      calcMode: 'dishPrice',
      dishPrice,
      serviceCharge: charge,
      guestCount: docData.guestCount || 0,
      standard: docData.standard || 0,
      roomName: docData.roomName || '',
      autoGenerated: true
    }

    if (hasExisting) {
      if (!isCreate) await db.updateDoc(COLLECTIONS.INCOME, first._id, incomeData)
    } else {
      await db.addDoc(COLLECTIONS.INCOME, incomeData)
      await db.updateDoc(COLLECTIONS.RESERVATION, reservationId, { hasIncome: true })
    }
  } catch (err) {
    console.warn('[banquet-sync] 同步收入失败:', err)
  }
},
```

```javascript
  async loadAvailableReservations() {
    try {
      // Load settings to get serviceChargeEnabledDate
      const settingsRes = await db.queryAll(COLLECTIONS.SETTINGS, {})
      const settings = {}
      ;(settingsRes.data || []).forEach(s => { settings[s.key] = s.value })
      const enabledDate = settings.serviceChargeEnabledDate

      if (!enabledDate) {
        this.setData({ recentReservations: [], pickerItems: [], pickerIndex: -1 })
        return
      }

      // Find confirmed reservations from enabledDate to today, within 30 days
      const now = new Date()
      const todayStr = formatDate(now)
      const thirtyDaysAgo = new Date(now.getTime() - 30 * 86400000)

      const startDate = enabledDate > formatDate(thirtyDaysAgo) ? new Date(enabledDate + 'T00:00:00') : thirtyDaysAgo
      const endDate = new Date(todayStr + 'T23:59:59')

      const _db = db.getDb()
      const _ = _db.command

      const resvRes = await db.queryAll(COLLECTIONS.RESERVATION, {
        date: _.gte(startDate).and(_.lte(endDate)),
        status: 'confirmed'
      })

      const allReservations = resvRes.data || []

      // Filter out ones that already have a linked banquet purchase
      const available = []
      for (const r of allReservations) {
        const linked = await db.queryAll(COLLECTIONS.PURCHASE, {
          sourceReservationId: r._id,
          category: 'banquet'
        })
        if (!linked.data || linked.data.length === 0) {
          available.push(r)
        }
      }

      // Sort by date descending
      available.sort((a, b) => {
        const dateA = a.date ? new Date(a.date).getTime() : 0
        const dateB = b.date ? new Date(b.date).getTime() : 0
        return dateB - dateA
      })

      // Build picker display strings
      const items = available.map(r => {
        const dateStr = formatDate(r.date)
        return dateStr + ' ' + (r.customerName || '') + ' ' + (r.time || '') + ' ' + (r.roomName || '')
      })

      this.setData({ recentReservations: available, pickerItems: items, pickerIndex: -1, selectedReservation: null })
    } catch (err) {
      console.warn('[purchase-add] 加载可用预约失败:', err)
    }
  },
```
